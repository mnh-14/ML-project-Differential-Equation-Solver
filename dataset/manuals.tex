\section*{Family: Linear}
\textbf{Question:} $5 y{\left(x \right)} \cos{\left(x \right)} + \frac{d}{d x} y{\left(x \right)} = 3 \sin{\left(x \right)}$

\textbf{Steps:}
\begin{enumerate}
\item identify_family :  $First-Order Linear$
\item find_P_Q :  $P(x)=5 \cos{\left(x \right)}, Q(x)=3 \sin{\left(x \right)}$
\item calc_mu :  $\mu(x) = e^{\int 5 \cos{\left(x \right)} dx} = e^{5 \sin{\left(x \right)}}$
\item integrate_product :  $\mu y = \int 3 e^{5 \sin{\left(x \right)}} \sin{\left(x \right)} dx$
\end{enumerate}
\textbf{Final Result:} $\left(C_{1} + 3 \int e^{5 \sin{\left(x \right)}} \sin{\left(x \right)}\, dx\right) e^{- 5 \sin{\left(x \right)}}$
