\section*{Family: Linear}
\textbf{Question:} $5 y{\left(x \right)} e^{x} + \frac{d}{d x} y{\left(x \right)} = 4 x$

\textbf{Steps:}
\begin{enumerate}
\item identify family :  $First-Order Linear$
\item find P Q :  $P(x)=5 e^{x}, Q(x)=4 x$
\item calc mu :  $\mu(x) = e^{\int 5 e^{x} dx} = e^{5 e^{x}}$
\item integrate product :  $\mu y = \int 4 x e^{5 e^{x}} dx$
\end{enumerate}
\textbf{Final Result:} $\left(C_{1} + 4 \int x e^{5 e^{x}}\, dx\right) e^{- 5 e^{x}}$
