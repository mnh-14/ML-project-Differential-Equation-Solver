\section*{Family: Linear}
\textbf{Question:} $\frac{d}{d x} y{\left(x \right)} + \frac{y{\left(x \right)}}{x} = \cos{\left(x \right)}$

\textbf{Steps:}
\begin{enumerate}
\item $Find Integrating Factor$ :  $\mu(x) = e^{\int \frac{1}{x} dx} = x$
\item $Multiply ODE by \mu(x)$ :  $x(y' + \frac{1}{x}y) = x \cos{\left(x \right)}$
\end{enumerate}
\textbf{Final Result:} $y{\left(x \right)} = \frac{C_{1}}{x} + \sin{\left(x \right)} + \frac{\cos{\left(x \right)}}{x}$
