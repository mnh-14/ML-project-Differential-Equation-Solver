\textbf{Question:} $5 x^{2} y{\left(x \right)} + \frac{d}{d x} y{\left(x \right)} = 3 x^{2}$

\textbf{Steps:}
\begin{enumerate}
\item Identify :  $P(x) = 5 x^{2}$
\item Int Factor Setup :  $\mu(x) = e^{\int \left5 x^{2} \right dx}$
\item Int Factor Calc :  $\mu(x) = e^{\frac{5 x^{3}}{3}}$
\item Multiply :  $\frac{d}{dx}(e^{\frac{5 x^{3}}{3}}y) = 3 x^{2} e^{\frac{5 x^{3}}{3}}$
\item Integrate :  $e^{\frac{5 x^{3}}{3}}y = \frac{3 e^{\frac{5 x^{3}}{3}}}{5} + C_1$
\end{enumerate}
\textbf{Final Result:} $y{\left(x \right)} = \left(C_{1} + \frac{3 e^{\frac{5 x^{3}}{3}}}{5}\right) e^{- \frac{5 x^{3}}{3}}$
