\textbf{Steps:}
\begin{enumerate}
\item Identify :  $P(x) = 4 \cos{\left(x \right)}$
\item Int Factor_Setup :  $\mu(x) = e^{\int 4 \cos{\left(x \right)} dx}$
\item Int Factor_Calc :  $\mu(x) = e^{4 \sin{\left(x \right)}}$
\item Multiply :  $\frac{d}{dx}(e^{4 \sin{\left(x \right)}}y) = e^{4 \sin{\left(x \right)}} \tan{\left(x \right)}$
\item Integrate :  $e^{4 \sin{\left(x \right)}}y = \int e^{4 \sin{\left(x \right)}} \tan{\left(x \right)}\, dx + C_1$
\end{enumerate}
\textbf{Final Result:} $y{\left(x \right)} = \left(C_{1} + \int e^{4 \sin{\left(x \right)}} \tan{\left(x \right)}\, dx\right) e^{- 4 \sin{\left(x \right)}}$
