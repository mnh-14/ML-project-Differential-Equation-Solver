\textbf{Steps:}
\begin{enumerate}
\item Identify :  $M(x, y) = 2 \sin{\left(x \right)} + \cos{\left(y \right)} + 6$
\item Identify :  $N(x, y) = - x \sin{\left(y \right)} + 3 y^{2}$
\item Integrate M(x, y) wrt x, [F(x, y)]: :  $x \cos{\left(y \right)} + 6 x - 2 \cos{\left(x \right)} + g(y)$
\item Differentiate F(x, y) wrt y :  :  $- x \sin{\left(y \right)} + g'(y)$
\item Solve N = diff(F) for g'(y) :  $3 y^{2}$
\item Integrate g'(y) wrt y :  :  $y^{3} + C$
\end{enumerate}
\textbf{Final Result:} $x \cos{\left(y \right)} + 6 x + y^{3} - 2 \cos{\left(x \right)} = C_{1}$
