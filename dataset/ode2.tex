\textbf{Question:} $2 y{\left(x \right)} \cos{\left(x \right)} + \frac{d}{d x} y{\left(x \right)} = 2 x$

\textbf{Steps:}
\begin{enumerate}
\item Identify :  $P(x) = 2 \cos{\left(x \right)}$
\item Int_Factor_Setup :  $\mu(x) = e^{\int 2 \cos{\left(x \right)} dx}$
\item Int_Factor_Calc :  $\mu(x) = e^{2 \sin{\left(x \right)}}$
\item Multiply :  $\frac{d}{dx}(e^{2 \sin{\left(x \right)}}y) = 2 x e^{2 \sin{\left(x \right)}}$
\item Integrate :  $e^{2 \sin{\left(x \right)}}y = 2 \int x e^{2 \sin{\left(x \right)}}\, dx + C_1$
\end{enumerate}
\textbf{Final Result:} $y{\left(x \right)} = \left(C_{1} + 2 \int x e^{2 \sin{\left(x \right)}}\, dx\right) e^{- 2 \sin{\left(x \right)}}$
